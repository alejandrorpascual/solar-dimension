// QA round 4 witness probe for #53. Prints the cables and fuse verdicts for each witness; asserts nothing.
// QA53_REV picks the quote-items.ts revision: head (default) or a copy of an older one (__qa53_items_<rev>.ts).
import { test } from 'vitest'
import { extraction, line, SAMPLE_KIT as KIT } from './quote-fixtures'
import type { QuoteLine } from './quote-schema'

const rev = process.env.QA53_REV ?? 'head'
const mod = rev === 'head' ? await import('./quote-items') : await import(`./__qa53_items_${rev}.ts`)
type Id = 'cables' | 'fuse'

const named = (text: string, awg: string | null = null, qty = 1) =>
  line('wiring', 1, { qty, description: text, source: { page: 1, text }, specs: { awg } })
/** A wiring line split from a bundle: its own description, the bundle's words as its source. */
const split = (description: string, bundle: string, awg: string | null) =>
  line('wiring', 0, { description, source: { page: 1, text: bundle }, specs: { awg } })
const fuse = (fuseAmps: number) => line('fuse', 1, { specs: { fuseAmps, fuseClass: 'T' } })
const verdict = (lines: QuoteLine[], id: Id) => {
  const c = mod.checkItems(extraction(lines), KIT).find((x: { id: string }) => x.id === id)!
  const f = c.facts as Record<string, unknown>
  const bits = [f.gauge, f.amps === undefined ? undefined : `${f.amps} A`, f.batteryCableUnstated ? 'unstated' : undefined, f.sizeUnknown ? 'size' : undefined]
  return `${c.status} ${bits.filter(Boolean).join(' ')}`.trim()
}

test('qa53 witnesses', () => {
  const rows = witnesses.map(([ac, name, lines, id, want]) =>
    `${ac.padEnd(4)}| ${name.padEnd(58)}| want: ${want.padEnd(18)}| got: ${verdict(lines, id)}`)
  console.log(`\n[${rev}]\n${rows.join('\n')}\n`)
})

// The words the Main cables check shows on the page (quote-text.ts), at head only: quote-text imports ./quote-items.
test('qa53 page words at head', async () => {
  if (rev !== 'head') return
  const { en } = await import('../i18n/en')
  const { es } = await import('../i18n/es')
  const { itemText, markLabel } = await import('./quote-text')
  const cases: Array<[string, QuoteLine[]]> = [
    ['AC1 split es: alimentador CA 4/0 + baterías 6', [split('Alimentador CA 4/0 AWG', WIRING_ES, '4/0'), split('Cable de baterías 6 AWG', WIRING_ES, '6')]],
    ['AC2 split es from a kit naming baterías', [split('Cable solar 10 AWG, 30 m', KIT_ES, '10')]],
    ['AC3 split: solar 10 + battery 4/0 from one kit line', [split('Cable solar 10 AWG', KIT_SOLAR_BATT, '10'), split('Cable de baterías 4/0 AWG', KIT_SOLAR_BATT, '4/0')]],
    ['AC4 split: solar 10 + unsized battery cable', [split('Cable solar 10 AWG', KIT_UNSIZED, '10'), split('Cableado de baterías', KIT_UNSIZED, null)]],
  ]
  const rows = cases.map(([name, lines]) => {
    const c = mod.checkItems(extraction(lines), KIT).find((x: { id: string }) => x.id === 'cables')!
    const say = (t: typeof en) => { const x = itemText(t, c, KIT, 'USD'); return `${markLabel(t, c)} · ${x.label} · ${x.detail ?? '-'}` }
    return `${name.padEnd(52)}| en: ${say(en).padEnd(48)}| es: ${say(es)}`
  })
  console.log(`\nPAGE\n${rows.join('\n')}\n`)
})
