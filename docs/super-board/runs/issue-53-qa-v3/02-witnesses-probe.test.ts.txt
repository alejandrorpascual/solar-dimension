// QA round 3 witness probe for #53. Prints the cables and fuse verdicts for each witness; asserts nothing.
import { test } from 'vitest'
import { extraction, line, SAMPLE_KIT as KIT } from './quote-fixtures'
import { checkItems, type ItemId } from './quote-items'
import type { QuoteLine } from './quote-schema'

const named = (text: string, awg: string | null = null, qty = 1) =>
  line('wiring', 1, { qty, description: text, source: { page: 1, text }, specs: { awg } })
/** A wiring line split from a bundle: its own description, the bundle's words as its source. */
const split = (description: string, bundle: string, awg: string | null) =>
  line('wiring', 0, { description, source: { page: 1, text: bundle }, specs: { awg } })
const fuse = (fuseAmps: number) => line('fuse', 1, { specs: { fuseAmps, fuseClass: 'T' } })
const verdict = (lines: QuoteLine[], id: ItemId) => {
  const c = checkItems(extraction(lines), KIT).find((x) => x.id === id)!
  const f = c.facts as Record<string, unknown>
  return `${c.status} ${JSON.stringify({ gauge: f.gauge, amps: f.amps, batteryCableUnstated: f.batteryCableUnstated, sizeUnknown: f.sizeUnknown })}`
}

test('qa53 witnesses', () => {
  const rows = witnesses.map(([ac, name, lines, id, want]) => `${ac} | ${name} | want: ${want} | got: ${verdict(lines, id)}`)
