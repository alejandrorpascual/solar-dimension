def run():
    p=subprocess.run(['npx','vitest','run','src/lib/quote-items.test.ts','src/lib/quote-text.test.ts'],capture_output=True,text=True)
    out=re.sub(r'\x1b\[[0-9;]*m','',p.stdout+p.stderr)
    m=re.search(r'Tests\s+(?:(\d+) failed \| )?(\d+) passed',out)
    failed=[l.strip()[:110] for l in out.splitlines() if l.strip().startswith('×') or re.match(r'\s*×',l)]
    return p.returncode, (m.group(1) or '0') if m else '?', failed, out
for name,path,old,new in M:
    src=open(path).read()
    if old not in src:
        print(f'{name}: PATTERN NOT FOUND'); continue
    open(path,'w').write(src.replace(old,new,1))
    try:
        rc,nf,failed,out=run()
    finally:
        open(path,'w').write(src)
    status='KILLED' if rc!=0 else 'SURVIVED'
    print(f'{name}: {status} ({nf} failing)')
    if rc==0: pass
