const OUT = process.env.QA_OUT ?? '/dev/null'
const log = (s: string) => appendFileSync(OUT, s + '\n')

type W = [string, number | null, string][]
const named = (text: string, awg: string | null = null, qty = 1) =>
  line('wiring', 1, { qty, description: text, source: { page: 1, text }, specs: { awg } })
const fuse250 = line('fuse', 1, { specs: { fuseAmps: 250, fuseClass: 'T' } })

const show = (mod: typeof head, lines: ReturnType<typeof named>[], id: 'cables' | 'fuse') => {
  const c = mod.checkItems(extraction(lines), KIT as never, {}).find((i) => i.id === id)!
  const f = c.facts
  const facts = [f.gauge, f.batteryCableUnstated ? 'batteryCableUnstated' : '', f.sizeUnknown ? 'sizeUnknown' : '', f.runsUnstated ? 'runsUnstated' : '', f.maxA ? `maxA${f.maxA}` : ''].filter(Boolean).join(',')
  return `${c.status}${facts ? ` [${facts}]` : ''} cites ${c.sources.map((s) => `"${s.text}"`).join(' + ')}`
}

test('witnesses: base (staging) vs head, cables', () => {
  const rows = witnesses.map(([label, ls]) => `${label}\n    base: ${show(base as never, ls, 'cables')}\n    head: ${show(head, ls, 'cables')}`)
  log('\n=== CABLES ===\n' + rows.join('\n'))
})
test('witnesses: base vs head, fuse 250 A', () => {
  const rows = witnesses.map(([label, ls]) => `${label} + fuse 250A\n    base: ${show(base as never, [...ls, fuse250], 'fuse')}\n    head: ${show(head, [...ls, fuse250], 'fuse')}`)
  log('\n=== FUSE ===\n' + rows.join('\n'))
})
test('sample quotes unchanged: every item verdict, base vs head', () => {
  const out: string[] = []
  for (const [name, q] of Object.entries(SAMPLE_QUOTES)) {
    const a = JSON.stringify(base.checkItems(q as never, KIT as never, {}))
    const b = JSON.stringify(head.checkItems(q as never, KIT as never, {}))
    out.push(`${name}: ${a === b ? 'identical' : 'DIFFERENT'}`)
  }
  log('\n=== SAMPLE QUOTES ===\n' + out.join('\n'))
})
