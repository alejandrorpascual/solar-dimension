const only = process.argv[2]
const run = (files) => { try { return { ok: true, out: execSync(`npx vitest run ${files} 2>&1`, { encoding: 'utf8' }) } } catch (e) { return { ok: false, out: e.stdout } } }
const rows = []
try {
  for (const [name, find, repl] of M) {
    if (only && !name.startsWith(only + ' ')) continue

    if (n !== 1) { rows.push(`${name}: SKIP (pattern found ${n}x)`); continue }

    const d = run('src/lib/quote-tax-differential.test.ts')

    const tc = /AssertionError: ([^\n]*)/.exec(d.out)?.[1]

} finally { fs.writeFileSync(F, orig) }
