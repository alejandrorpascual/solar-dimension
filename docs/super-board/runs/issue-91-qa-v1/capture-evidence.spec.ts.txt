// QA evidence capture for issue #91: not part of the suite, never committed. QA_PHASE=before|after, QA_SHOTS_DIR=<dir>.
import { mkdirSync } from 'node:fs'
import { join } from 'node:path'
import { expect, test, type Page } from '@playwright/test'

const PHASE = process.env.QA_PHASE ?? 'after'
const DIR = process.env.QA_SHOTS_DIR ?? 'qa-shots'
mkdirSync(DIR, { recursive: true })
const STORAGE_KEY = 'solar-dimension.projects.v1'

const VIEWPORTS = [
  { name: 'desktop', width: 1920, height: 1080, crop: true },
  { name: 'tablet', width: 1024, height: 768, crop: true },
  { name: 'mobile', width: 375, height: 667, crop: false },
] as const
type Viewport = (typeof VIEWPORTS)[number]

test.use({ locale: 'es-VE' })

const firstHeading = (page: Page) => page.getByRole('heading', { name: '¿Qué aparatos deben seguir encendidos?' })

/** Opens the projects sheet when the cards sit behind the "Proyectos · N" button at this width. */
async function revealProjects(page: Page) {
  const open = page.getByRole('button', { name: /^Proyectos · \d+$/ })
  if (await open.isVisible()) {
    await open.click()
    await expect(page.getByRole('dialog')).toBeVisible()
  }
  await expect(page.locator('.pcards').filter({ visible: true })).toBeVisible()
}

async function shoot(page: Page, vp: Viewport, name: string, crop: ReturnType<Page['locator']> | null) {
  await page.screenshot({ path: join(DIR, `${PHASE}-${name}-${vp.name}.png`) })
  if (vp.crop && crop) await crop.screenshot({ path: join(DIR, `${PHASE}-${name}-${vp.name}-crop.png`) })
}

for (const vp of VIEWPORTS) {
  test(`${PHASE} · projects · ${vp.name}`, async ({ page }) => {
    await page.setViewportSize({ width: 1920, height: 1080 })
    await page.goto('/')
    await expect(firstHeading(page)).toBeVisible()
    const field = page.getByLabel('Nombre del proyecto')
    await expect(async () => {
      if (!(await field.isVisible())) await page.locator('.pcards__new').filter({ visible: true }).click()
      await expect(field).toBeVisible({ timeout: 1000 })
    }).toPass()
    await field.press('Enter')
    await page.locator('.pcard--on').getByRole('button', { name: /^Más acciones de / }).click()
    await page.getByRole('menu').getByRole('menuitem', { name: 'Duplicar' }).click()
    await expect(page.locator('.pcard')).toHaveCount(3)
    await page.setViewportSize({ width: vp.width, height: vp.height })
    await revealProjects(page)
    await page.waitForTimeout(400)
    await shoot(page, vp, 'ac3-projects', page.locator('.pcards').filter({ visible: true }))
  })

  test(`${PHASE} · custom loads · ${vp.name}`, async ({ page }) => {
    await page.addInitScript((key) => {
      if (localStorage.getItem(key)) return
      const saved = { id: 'custom1', name: 'Bomba vieja', watts: 180, surge: 180, note: 'Custom · 180 W' }
      const project = {
        id: 'p1', name: 'Garita', qty: { custom1: 1 }, custom: [saved], hours: 8, volts: 48, chemistry: 'lifepo4', batteryUnit: null, updatedAt: 1,
        dutyOverrides: {},
      }
      localStorage.setItem(key, JSON.stringify({ version: 1, projects: [project], currentId: 'p1' }))
    }, STORAGE_KEY)
    await page.setViewportSize({ width: vp.width, height: vp.height })
    await page.goto('/')
    await expect(firstHeading(page)).toBeVisible()
    await page.getByLabel('Nombre', { exact: true }).fill('Bomba de agua')
    await page.getByLabel('W en marcha').fill('1500')
    const added = page.locator('.card').filter({ hasText: 'Bomba de agua' })
    await expect(async () => {
      await page.getByRole('button', { name: 'Agregar aparato' }).click()
      await expect(added).toBeVisible({ timeout: 1000 })
    }).toPass()
    const section = page.locator('.catalog__section').filter({ hasText: 'Bomba de agua' })
    await section.scrollIntoViewIfNeeded()
    await page.waitForTimeout(400)
    await shoot(page, vp, 'ac4-custom', section)
  })

  test(`${PHASE} · sun hint · ${vp.name}`, async ({ page }) => {
    await page.setViewportSize({ width: vp.width, height: vp.height })
    await page.goto('/?step=3')
    const sun = page.locator('.planner__sun').filter({ visible: true })
    await expect(async () => {
      if (!(await sun.isVisible())) await page.getByRole('button', { name: /^Sí, agregar paneles/ }).click()
      await expect(sun).toBeVisible({ timeout: 1000 })
    }).toPass()
    await sun.scrollIntoViewIfNeeded()
    await page.waitForTimeout(400)
    await shoot(page, vp, 'ac1-sun-hint', sun)
  })
}

test(`${PHASE} · fine print · desktop`, async ({ page }) => {
  await page.setViewportSize({ width: 1920, height: 1080 })
  await page.goto('/?step=4')
  const fine = page.locator('p.fine').filter({ hasText: 'Supone' }).filter({ visible: true }).first()
  await expect(fine).toBeVisible()
  await fine.scrollIntoViewIfNeeded()
  await fine.screenshot({ path: join(DIR, `${PHASE}-ac2-fine-print-desktop-crop.png`) })
  console.log(`FINE PRINT (${PHASE}): ${await fine.textContent()}`)
})
