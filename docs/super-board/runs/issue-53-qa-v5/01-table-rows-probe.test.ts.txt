// QA round 5 witness probe for #53. Runs every wording of the ticket's table through the Main cables check and prints
// what the page shows (mark · label, or the detail of an item left to confirm), in English and Spanish. It asserts
// nothing; the last line counts the rows that differ from the table.
// QA53_REV picks the quote-items.ts revision: head (default) or a copy of an older one (__qa53_items_<rev>.ts).
import { test } from 'vitest'
import { en } from '../i18n/en'
import { es } from '../i18n/es'
import { extraction, line, SAMPLE_KIT as KIT } from './quote-fixtures'
import type { ItemCheck } from './quote-items'
import type { QuoteLine } from './quote-schema'
import { itemText, markLabel } from './quote-text'

const rev = process.env.QA53_REV ?? 'head'
const mod = rev === 'head' ? await import('./quote-items') : await import(`./__qa53_items_${rev}.ts`)
type Id = 'cables' | 'fuse'

const named = (text: string, awg: string | null = null, qty = 1) =>
  line('wiring', 1, { qty, description: text, source: { page: 1, text }, specs: { awg } })
/** A wiring line split from one quote line: its own description, the quote line's words as its source. */
const split = (description: string, bundle: string, awg: string | null = null) =>
  line('wiring', 0, { description, source: { page: 1, text: bundle }, specs: { awg } })
const fuse = (fuseAmps: number) => line('fuse', 1, { specs: { fuseAmps, fuseClass: 'T' } })

const judge = (lines: QuoteLine[], id: Id): ItemCheck => mod.checkItems(extraction(lines), KIT).find((x: ItemCheck) => x.id === id)!
/** What the page shows for one check: "Too small · 6 AWG", or the detail of an item left to confirm. */
const reads = (t: typeof en, c: ItemCheck): string => {
  const { label, detail } = itemText(t, c, KIT, 'USD')
  const mark = markLabel(t, c)
  return c.status === 'unknown' ? `${mark} · ${detail ?? label}` : `${mark} · ${label}`
}

const SOLAR10 = named('Cable solar 10 AWG', '10')

const THIN = named('Cable 6 AWG baterías', '6')
const WIRING = 'Cableado: alimentador CA 4/0 AWG y cable de baterías 6 AWG'
const KIT_BATT = 'Kit solar: 4 paneles 550 W, 2 baterías de litio 200 Ah, cable solar 10 AWG'
const KIT_GROUND = 'Kit solar: 4 paneles 550 W, 2 baterías de litio 200 Ah, cable solar 10 AWG, cable de tierra 8 AWG'
const kitSiblings = (bundle: string) => [line('panel', 0, { source: { page: 1, text: bundle } }), line('battery', 0, { source: { page: 1, text: bundle } })]

const SHORT6 = 'Too small · 6 AWG'
const UNSTATED = 'To confirm · Battery cable not stated'

/** [table row, wording, lines, check, what the table says the page shows] */

  ['5', 'Cable solar 10 AWG alone, own line', [SOLAR10], 'cables', UNSTATED],
  ['5', `Cable solar 10 AWG split from "${KIT_BATT}"`, [...kitSiblings(KIT_BATT), split('Cable solar 10 AWG', KIT_BATT, '10')], 'cables', UNSTATED],

test('qa53 v5 witnesses', () => {
  let differ = 0
  console.log(`revision: ${rev}`)
  for (const [row, wording, lines, id, want] of rows) {
    const c = judge(lines, id)
    const got = reads(en, c)
    const ok = got === want
    if (!ok) differ += 1
    console.log(`${ok ? 'as table' : 'DIFFERS '} | row ${row.padEnd(8)} | ${wording}\n           en: ${got}${ok ? '' : `   (table: ${want})`}\n           es: ${reads(es, c)}`)
  }
  console.log(`rows that differ from the table: ${differ} of ${rows.length}`)
})
