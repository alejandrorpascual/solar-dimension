# Accounts, Cloud Projects and Spanish Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a signed-in user keep their projects in Postgres across devices (anonymous use keeps working from local storage and is merged into the account on first sign-in), and make the whole app usable in Spanish.

**Architecture:** Neon Postgres via the Vercel Marketplace, Drizzle ORM with SQL migrations checked in, Better Auth (email + password) mounted as a TanStack Start server route, TanStack Start server functions for project CRUD scoped to the session user. The root route's `beforeLoad` resolves `{ user, locale }` once per request so the first render is already in the right language and signed-in state. Localization is a typed dictionary per language; the engine stops emitting prose (warnings become codes with params) and the UI renders text through the dictionary.

**Tech Stack:** existing app (TanStack Start, React 19, zod, Vitest, Playwright) plus `drizzle-orm`, `drizzle-kit`, `@neondatabase/serverless`, `better-auth`, `@better-auth/drizzle-adapter`, `dotenv`.

## Global Constraints

- Node 24, npm. TypeScript `strict: true`; `npm run typecheck` passes at the end of every task.
- Immutability everywhere (new objects, never in-place mutation). No `console.log`. Files under 400 lines, functions under 50.
- Coverage threshold stays 80% on `src/lib/**`; `src/i18n/**` is added to the coverage include.
- Secrets only from environment variables: `DATABASE_URL`, `BETTER_AUTH_SECRET`. Never commit `.env*` (already ignored). Fail fast with a clear error when a required variable is missing on the server.
- Every server function that touches the database first resolves the session and scopes every query by `userId`. Never trust a client-supplied user id.
- All user-visible strings come from `src/i18n/*`; no English literals in components after Task 3. Number formatting goes through `src/i18n/format.ts` with the active locale.
- Locales: `en` (default) and `es` (Latin American Spanish, `es-419` number formatting: `1,100 W`, `3.7 kWh`).
- Commit after every task with a conventional-commit message; no attribution lines. Do not push; the orchestrator pushes.
- The Playwright suite must keep passing; existing English assertions stay valid because `en` is the default locale.

- [ ] **Step 2: Engine warning codes**

In `src/lib/sizing.ts` replace `buildWarnings` and its call, add the types, and add `largestSurgeId`:

```ts
export type WarningCode = 'inverterVoltage' | 'highCurrent' | 'parallelStrings' | 'dominantLoad'
export interface WarningParams {
  inverterW?: number; volts?: number; voltsList?: string; amps?: number; nextVolts?: number | null
  strings?: number; itemId?: string; itemName?: string; sharePct?: number
}
export interface Warning { code: WarningCode; params: WarningParams }

function buildWarnings(ctx: WarningContext): Warning[] {
  if (ctx.items.length === 0) return []
  const w: Warning[] = []
  if (!ctx.inverterVoltsOk) {
    w.push({ code: 'inverterVoltage', params: { inverterW: ctx.inverter.w, voltsList: ctx.inverter.volts.join(' / ') } })
  }
  if (ctx.amps > SIZING.MAX_PRACTICAL_AMPS) {
    w.push({ code: 'highCurrent', params: { amps: Math.round(ctx.amps), volts: ctx.volts, nextVolts: VOLTAGES.find((v) => v > ctx.volts) ?? null } })
  }
  if (ctx.battery && ctx.battery.parallel > SIZING.MAX_PARALLEL_STRINGS) {
    w.push({ code: 'parallelStrings', params: { strings: ctx.battery.parallel } })
  }
  const dominant = ctx.items.find((i) => ctx.loadWh > 0 && i.wh / ctx.loadWh > SIZING.DOMINANT_SHARE)
  if (dominant) {
    w.push({ code: 'dominantLoad', params: { itemId: dominant.id, itemName: dominant.name, sharePct: Math.round((dominant.wh / ctx.loadWh) * 100) } })
  }
  return w
}
```

```ts
import { en } from './en'
import { es } from './es'
import type { Dictionary, Locale, UiKey } from './types'

export type { Dictionary, Locale, UiKey } from './types'
export const LOCALES: readonly Locale[] = ['en', 'es']
export const DEFAULT_LOCALE: Locale = 'en'
export const LOCALE_COOKIE = 'lang'

const DICTIONARIES: Record<Locale, Dictionary> = { en, es }

export function isLocale(v: unknown): v is Locale {
  return typeof v === 'string' && (LOCALES as readonly string[]).includes(v)
}

export function getDictionary(locale: Locale): Dictionary {
  return DICTIONARIES[locale]
}

function cookieValue(cookie: string | null | undefined, name: string): string | null {
  if (!cookie) return null
  const hit = cookie.split(';').map((c) => c.trim()).find((c) => c.startsWith(`${name}=`))
  return hit ? decodeURIComponent(hit.slice(name.length + 1)) : null
}

export function resolveLocale(input: { cookie?: string | null; acceptLanguage?: string | null }): Locale {
  const fromCookie = cookieValue(input.cookie, LOCALE_COOKIE)
  if (isLocale(fromCookie)) return fromCookie
  const first = (input.acceptLanguage ?? '').split(',')[0]?.trim().toLowerCase().slice(0, 2)
  return isLocale(first) ? first : DEFAULT_LOCALE
}

/** Replace `{key}` placeholders in a ui string. */
export function tr(t: Dictionary, key: UiKey, params: Record<string, string | number> = {}): string {
  return t.ui[key].replace(/\{(\w+)\}/g, (_, k: string) => String(params[k] ?? ''))
}

export function applianceName(t: Dictionary, item: { id: string; name: string; isCustom?: boolean }): string {
  return item.isCustom ? item.name : (t.appliances[item.id]?.name ?? item.name)
}

export function applianceNote(t: Dictionary, item: { id: string; note: string; isCustom?: boolean }): string {
  return item.isCustom ? item.note : (t.appliances[item.id]?.note ?? item.note)
}
```

```ts
import type { Locale } from './types'

const TAGS: Record<Locale, string> = { en: 'en-US', es: 'es-419' }
const KW_THRESHOLD = 10000

export function fmtInt(n: number, locale: Locale): string {
  return Math.round(n).toLocaleString(TAGS[locale])
}
export function fmtW(w: number, locale: Locale): string {
  return w >= KW_THRESHOLD ? `${(w / 1000).toLocaleString(TAGS[locale], { minimumFractionDigits: 1, maximumFractionDigits: 1 })} kW` : `${fmtInt(w, locale)} W`
}
export function fmtKwh(wh: number, locale: Locale): string {
  return `${(wh / 1000).toLocaleString(TAGS[locale], { minimumFractionDigits: 1, maximumFractionDigits: 1 })} kWh`
}
export function fmtHours(h: number, locale: Locale): string {
  return `${(Math.round(h * 10) / 10).toLocaleString(TAGS[locale])} h`
}
```

```tsx
const LABELS: Record<Locale, string> = { en: 'English', es: 'Español' }

export function LanguageSwitch() {
  const { locale, t } = useLocale()
  const router = useRouter()
  async function pick(next: Locale) {

    </div>
  )
}
```

```ts
import type { Project } from './schema'

export function diffProjects(prev: readonly Project[], next: readonly Project[]): { upserts: Project[]; deletes: string[] } {
  const before = new Map(prev.map((p) => [p.id, JSON.stringify(p)]))
  const after = new Set(next.map((p) => p.id))
  const upserts = next.filter((p) => before.get(p.id) !== JSON.stringify(p))
  const deletes = prev.filter((p) => !after.has(p.id)).map((p) => p.id)
  return { upserts, deletes }
}
```

```ts
import { createServerFn } from '@tanstack/react-start'
import { and, eq, inArray } from 'drizzle-orm'
import { z } from 'zod'
import { db } from '../db/client'
import { projects } from '../db/schema'
import { ProjectSchema, type Project } from '../lib/schema'
import { requireUserId } from './session'

const ProjectsInput = z.object({ projects: z.array(ProjectSchema).max(200) })
const IdsInput = z.object({ ids: z.array(z.string().min(1)).max(200) })

function toProject(row: typeof projects.$inferSelect): Project {
  return ProjectSchema.parse({ id: row.id, name: row.name, qty: row.qty, custom: row.custom, hours: row.hours, volts: row.volts, chemistry: row.chemistry })
}

async function upsertAll(userId: string, list: Project[]): Promise<void> {
  if (list.length === 0) return
  await db
    .insert(projects)
    .values(list.map((p) => ({ ...p, userId, updatedAt: new Date() })))
    .onConflictDoUpdate({
      target: [projects.userId, projects.id],
      set: { name: projects.name, qty: projects.qty, custom: projects.custom, hours: projects.hours, volts: projects.volts, chemistry: projects.chemistry, updatedAt: new Date() },
    })
}

export const listProjects = createServerFn({ method: 'GET' }).handler(async (): Promise<Project[]> => {
  const userId = await requireUserId()
  const rows = await db.select().from(projects).where(eq(projects.userId, userId)).orderBy(projects.createdAt)
  return rows.map(toProject)
})

export const saveProjects = createServerFn({ method: 'POST' })
  .validator((d: unknown) => ProjectsInput.parse(d))
  .handler(async ({ data }) => { await upsertAll(await requireUserId(), data.projects); return { ok: true } })

export const deleteProjects = createServerFn({ method: 'POST' })
  .validator((d: unknown) => IdsInput.parse(d))
  .handler(async ({ data }) => {
    const userId = await requireUserId()
    if (data.ids.length) await db.delete(projects).where(and(eq(projects.userId, userId), inArray(projects.id, data.ids)))
    return { ok: true }
  })

export const importProjects = createServerFn({ method: 'POST' })
  .validator((d: unknown) => ProjectsInput.parse(d))
  .handler(async ({ data }): Promise<Project[]> => {
    const userId = await requireUserId()
    await upsertAll(userId, data.projects)
    const rows = await db.select().from(projects).where(eq(projects.userId, userId)).orderBy(projects.createdAt)
    return rows.map(toProject)
  })
```

Note on the `onConflictDoUpdate` `set`: Drizzle needs `sql\`excluded.name\`` style references to take the incoming values. Use `import { sql } from 'drizzle-orm'` and `set: { name: sql\`excluded.name\`, qty: sql\`excluded.qty\`, custom: sql\`excluded.custom\`, hours: sql\`excluded.hours\`, volts: sql\`excluded.volts\`, chemistry: sql\`excluded.chemistry\`, updatedAt: new Date() }`.

```ts
import { useRouteContext } from '@tanstack/react-router'
import { useCallback, useEffect, useRef, useState } from 'react'
import { loadStore, saveStore, SYNCED_KEY } from '../lib/persistence'
import type { Project, Store } from '../lib/schema'
import { currentProject, initialStore } from '../lib/store'
import { diffProjects } from '../lib/sync'
import { deleteProjects, importProjects, listProjects, saveProjects } from '../server/projects'

function fromServer(list: Project[], prev: Store): Store {
  const projects = list.length ? list : prev.projects
  const currentId = projects.some((p) => p.id === prev.currentId) ? prev.currentId : projects[0]!.id
  return { version: 1, projects, currentId }
}

export function useProjectStore() {
  const { user } = useRouteContext({ from: '__root__' })
  const [store, setStore] = useState<Store>(initialStore)
  const [syncing, setSyncing] = useState(false)
  const [syncError, setSyncError] = useState(false)
  const latest = useRef(store)
  latest.current = store

  useEffect(() => {
    const local = loadStore() ?? initialStore()
    setStore(local)
    if (!user) return
    let cancelled = false
    setSyncing(true)
    const alreadySynced = safeGet(SYNCED_KEY) === user.id
    const load = alreadySynced ? listProjects() : importProjects({ data: { projects: local.projects } })
    load
      .then((list) => {
        if (cancelled) return
        safeSet(SYNCED_KEY, user.id)
        const next = fromServer(list, local)
        setStore(next)
        saveStore(next)
      })
      .catch(() => { if (!cancelled) setSyncError(true) })
      .finally(() => { if (!cancelled) setSyncing(false) })
    return () => { cancelled = true }
  }, [user?.id])

  const commit = useCallback((next: Store) => {
    const prev = latest.current
    setStore(next)
    saveStore(next)
    if (!user) return
    const { upserts, deletes } = diffProjects(prev.projects, next.projects)
    const ops: Promise<unknown>[] = []
    if (upserts.length) ops.push(saveProjects({ data: { projects: upserts } }))
    if (deletes.length) ops.push(deleteProjects({ data: { ids: deletes } }))
    Promise.all(ops).then(() => setSyncError(false)).catch(() => setSyncError(true))
  }, [user?.id])

  return { store, project: currentProject(store), commit, syncing, syncError }
}

function safeGet(key: string): string | null { try { return localStorage.getItem(key) } catch { return null } }
function safeSet(key: string, v: string): void { try { localStorage.setItem(key, v) } catch { /* ignore */ } }
```
