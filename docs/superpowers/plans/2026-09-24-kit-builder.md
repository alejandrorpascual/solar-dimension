# Kit Builder Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the "Kit builder" (Variant C of the approved design) as a deployable TanStack Start web app: a three-step wizard that turns a list of appliances into an inverter, a battery bank and a DC cabling list, with named projects persisted in the browser.

**Architecture:** A pure TypeScript sizing engine under `src/lib/` (catalog, sizing math, custom-load parsing, project store, persistence) with unit tests, consumed by a single TanStack Router route (`/`) whose `step` search param drives three screens built from small React components. No backend: projects live in `localStorage` behind a zod-validated, versioned schema. Deployed to Vercel through Nitro.

**Tech Stack:** TanStack Start (React 19, Vite, Nitro), TypeScript strict, zod, lucide-react, plain CSS with custom properties, Vitest (+ v8 coverage), Playwright, GitHub Actions, Vercel.

**Design reference:** the approved Design artifact, Variant C "Kit builder" (desktop 1280 wide and phone 390 wide artboards): https://claude.ai/artifact/1h6VexfaNCoyWuwtfJhorF . Fonts: Bricolage Grotesque (display) + Figtree (body). Palette: ground `#F2F4F7`, surface `#FFFFFF`, ink `#12161C`, muted `#5C6470`, line `#DDE2E9`, accent `#2447E6`, accent-tint `#EEF2FF`, warn-bg `#FFF4E0`, warn-border `#F0C36B`, warn-ink `#6B4A00`, error `#B3261E`.

## Global Constraints

- Node 24, npm (lockfile `package-lock.json`); no other package manager.
- TypeScript `strict: true`; `npm run typecheck` must pass at the end of every task.
- Every `src/lib/*.ts` module is pure (no DOM, no React) except `src/lib/persistence.ts`.
- Immutability: never mutate a store, project or array in place; return new objects.
- No `console.log` in committed code.
- Files under 400 lines; functions under 50 lines.
- Coverage threshold 80% (lines, functions, statements, branches) on `src/lib/**`, enforced by `npm run test:coverage`.
- Sizing constants live in `src/lib/catalog.ts` and `src/lib/sizing.ts`, never inline in components.
- Copy rules: appliance names use `·` (middle dot) as the separator in AC names (`2 ton AC · inverter`); watts formatted as `1,100 W`, kilowatts above 10 kW as `12.0 kW`, energy as `3.7 kWh`, hours as `20 h`.
- Commit after every task with a conventional-commit message (`feat:`, `test:`, `chore:`), no attribution lines.
- Touch targets in the UI are at least 44 px tall on phone widths.

```ts
const KW_THRESHOLD = 10000

export function fmtInt(n: number): string {
  return Math.round(n).toLocaleString('en-US')
}

export function fmtW(w: number): string {
  return w >= KW_THRESHOLD ? `${(w / 1000).toFixed(1)} kW` : `${fmtInt(w)} W`
}

export function fmtKwh(wh: number): string {
  return `${(wh / 1000).toFixed(1)} kWh`
}

export function fmtHours(h: number): string {
  return `${Math.round(h * 10) / 10} h`
}
```

```ts
import {
  APPLIANCES, CHEMISTRIES, FUSES, GAUGES, INVERTERS, SIZING, VOLTAGES,
  type CatalogItem, type Chemistry, type ChemistryId, type Inverter, type Voltage,
} from './catalog'
import { fmtInt, fmtW } from './format'

export interface CustomLoadLike { id: string; name: string; watts: number; surge: number; note: string }

export interface SizingInput {
  qty: Record<string, number>
  custom?: readonly CustomLoadLike[]
  hours: number
  volts: Voltage
  chemistry: ChemistryId
  runLengthM?: number
}

export interface SelectedItem extends CatalogItem { qty: number; totalWatts: number; wh: number }
export interface InverterPick { w: number; surge: number; volts: readonly Voltage[]; voltsOk: boolean }
export interface BatteryPick {
  unitLabel: string; unitV: number; unitAh: number; unitWh: number
  series: number; parallel: number; count: number; config: string
  bankWh: number; usableWh: number; bankAh: number; hoursActual: number
}
export interface CablePick {
  amps: number; perRun: number; runs: number; gauge: string; mm2: number; ampacity: number
  mainCount: number; mainLengthM: number; seriesLinks: number; parallelLinks: number; linkLengthM: number
  totalLengthM: number; fuseA: number; dropPct: number
}
export interface SizingResult {
  empty: boolean; items: SelectedItem[]; count: number
  hours: number; volts: Voltage; chemistry: ChemistryId; chemistryLabel: string
  runningW: number; surgeW: number; largestSurgeName: string
  loadWh: number; requiredWh: number; requiredAh: number
  inverter: InverterPick; battery: BatteryPick | null; cable: CablePick; warnings: string[]
}

const HOURS_RANGE = [1, 72] as const
const RUN_LENGTH_RANGE = [0.5, 10] as const
const DEFAULT_RUN_LENGTH_M = 1.5
const FUSE_MARGIN = 1.1

const clamp = (n: number, lo: number, hi: number) => Math.min(hi, Math.max(lo, n))
const finite = (v: unknown, fallback: number) => (typeof v === 'number' && Number.isFinite(v) ? v : fallback)
const sum = <T>(list: readonly T[], f: (x: T) => number) => list.reduce((a, x) => a + f(x), 0)

export function catalog(custom: readonly CustomLoadLike[] = []): CatalogItem[] {
  const extra: CatalogItem[] = custom.map((c) => ({
    id: c.id, name: c.name, cat: 'Custom', watts: c.watts, surge: c.surge, duty: 1, note: c.note, isCustom: true,
  }))
  return [...APPLIANCES, ...extra]
}

function selectItems(items: readonly CatalogItem[], qty: Record<string, number>, hours: number): SelectedItem[] {
  return items
    .filter((a) => (qty[a.id] ?? 0) > 0)
    .map((a) => {
      const n = Math.trunc(qty[a.id] ?? 0)
      return { ...a, qty: n, totalWatts: a.watts * n, wh: Math.round(a.watts * a.duty * hours * n) }
    })
}

function pickInverter(runningW: number, surgeW: number, volts: Voltage): Inverter {
  const needW = runningW * SIZING.HEADROOM
  const fits = (inv: Inverter) => inv.w >= needW && inv.surge >= surgeW
  return (
    INVERTERS.find((inv) => inv.volts.includes(volts) && fits(inv)) ??
    INVERTERS.find(fits) ??
    INVERTERS[INVERTERS.length - 1]!
  )
}

function pickBattery(chem: Chemistry, volts: Voltage, requiredAh: number): BatteryPick | null {
  const candidates = chem.units
    .filter((u) => volts % u.v === 0)
    .map((u) => {
      const series = volts / u.v
      const parallel = Math.max(1, Math.ceil(requiredAh / u.ah))
      const count = series * parallel
      const bankWh = u.v * u.ah * count
      return {
        unitLabel: `${u.v} V ${u.ah} Ah ${chem.short}`, unitV: u.v, unitAh: u.ah, unitWh: u.v * u.ah,
        series, parallel, count, config: `${series}S${parallel}P`,
        bankWh, usableWh: Math.round(bankWh * chem.dod), bankAh: u.ah * parallel, hoursActual: 0,
      }
    })
    .sort((a, b) => a.count - b.count || a.bankWh - b.bankWh)
  return candidates[0] ?? null
}

function sizeCable(inverterW: number, volts: Voltage, runLengthM: number) {
  const amps = inverterW / volts / SIZING.INVERTER_EFF
  const top = GAUGES[GAUGES.length - 1]!
  let runs = 1
  while (amps / runs > top.amps) runs++
  const perRun = amps / runs
  const dropPct = (ohmPerM: number) => ((2 * runLengthM * perRun * ohmPerM) / volts) * 100
  let gi = GAUGES.findIndex((g) => g.amps >= perRun)
  if (gi < 0) gi = GAUGES.length - 1
  while (gi < GAUGES.length - 1 && dropPct(GAUGES[gi]!.ohmPerM) > SIZING.MAX_DROP_PCT) gi++
  const fuseA = FUSES.find((f) => f >= perRun * FUSE_MARGIN) ?? FUSES[FUSES.length - 1]!
  while (gi < GAUGES.length - 1 && fuseA > GAUGES[gi]!.amps) gi++
  const g = GAUGES[gi]!
  return { amps, perRun, runs, gauge: g.awg, mm2: g.mm2, ampacity: g.amps, fuseA: Math.min(fuseA, g.amps), dropPct: dropPct(g.ohmPerM) }
}

interface WarningContext {
  items: SelectedItem[]; inverter: Inverter; inverterVoltsOk: boolean
  amps: number; volts: Voltage; battery: BatteryPick | null; loadWh: number
}

function buildWarnings(ctx: WarningContext): string[] {
  if (ctx.items.length === 0) return []
  const w: string[] = []
  if (!ctx.inverterVoltsOk) {
    w.push(`${fmtW(ctx.inverter.w)} inverters are normally built for ${ctx.inverter.volts.join(' / ')} V. Change the battery voltage or trim the load.`)
  }
  if (ctx.amps > SIZING.MAX_PRACTICAL_AMPS) {
    const next = VOLTAGES.find((v) => v > ctx.volts)
    w.push(`${fmtInt(ctx.amps)} A on the DC side at ${ctx.volts} V is hard to wire safely${next ? `. Consider ${next} V.` : '.'}`)
  }
  if (ctx.battery && ctx.battery.parallel > SIZING.MAX_PARALLEL_STRINGS) {
    w.push(`${ctx.battery.parallel} parallel strings drift out of balance. Prefer bigger units or a higher voltage.`)
  }
  const dominant = ctx.items.find((i) => ctx.loadWh > 0 && i.wh / ctx.loadWh > SIZING.DOMINANT_SHARE)
  if (dominant) {
    w.push(`${dominant.name} is ${Math.round((dominant.wh / ctx.loadWh) * 100)}% of the energy. Dropping it is the cheapest way to shrink the bank.`)
  }
  return w
}

export function compute(input: SizingInput): SizingResult {
  const hours = clamp(finite(input.hours, 10), HOURS_RANGE[0], HOURS_RANGE[1])
  const volts: Voltage = VOLTAGES.includes(input.volts) ? input.volts : 48
  const chem = CHEMISTRIES[input.chemistry] ?? CHEMISTRIES.lifepo4
  const runLengthM = clamp(finite(input.runLengthM, DEFAULT_RUN_LENGTH_M), RUN_LENGTH_RANGE[0], RUN_LENGTH_RANGE[1])

  const items = selectItems(catalog(input.custom), input.qty ?? {}, hours)
  const runningW = sum(items, (i) => i.totalWatts)
  const largest = items.reduce<SelectedItem | null>(
    (best, i) => (i.surge - i.watts > (best ? best.surge - best.watts : -1) ? i : best), null,
  )
  const surgeW = runningW + (largest ? largest.surge - largest.watts : 0)
  const loadWh = sum(items, (i) => i.wh)
  const requiredWh = loadWh / SIZING.INVERTER_EFF / chem.dod
  const requiredAh = requiredWh / volts

  const inverter = pickInverter(runningW, surgeW, volts)
  const inverterVoltsOk = inverter.volts.includes(volts)
  const picked = pickBattery(chem, volts, requiredAh)
  const battery = picked
    ? { ...picked, hoursActual: loadWh > 0 ? (picked.bankWh * chem.dod * SIZING.INVERTER_EFF) / (loadWh / hours) : 0 }
    : null
  const cable = sizeCable(inverter.w, volts, runLengthM)
  const seriesLinks = battery ? (battery.series - 1) * battery.parallel : 0
  const parallelLinks = battery ? 2 * (battery.parallel - 1) : 0
  const linkLengthM = SIZING.LINK_LENGTH_M

  return {
    empty: items.length === 0,
    items,
    count: sum(items, (i) => i.qty),
    hours, volts, chemistry: chem.id, chemistryLabel: chem.label,
    runningW, surgeW, largestSurgeName: largest?.name ?? '',
    loadWh, requiredWh: Math.round(requiredWh), requiredAh: Math.round(requiredAh),
    inverter: { w: inverter.w, surge: inverter.surge, volts: inverter.volts, voltsOk: inverterVoltsOk },
    battery,
    cable: {
      amps: Math.round(cable.amps), perRun: Math.round(cable.perRun), runs: cable.runs,
      gauge: cable.gauge, mm2: cable.mm2, ampacity: cable.ampacity,
      mainCount: 2 * cable.runs, mainLengthM: runLengthM,
      seriesLinks, parallelLinks, linkLengthM,
      totalLengthM: Math.round((2 * cable.runs * runLengthM + (seriesLinks + parallelLinks) * linkLengthM) * 10) / 10,
      fuseA: cable.fuseA, dropPct: Math.round(cable.dropPct * 10) / 10,
    },
    warnings: buildWarnings({ items, inverter, inverterVoltsOk, amps: cable.amps, volts, battery, loadWh }),
  }
}
```

```ts
export interface CustomForm {
  name: string
  mode: 'watts' | 'va'
  watts: string
  volts: string
  amps: string
  surge: string
}

export const EMPTY_FORM: CustomForm = { name: '', mode: 'watts', watts: '', volts: '', amps: '', surge: '' }

export interface ParsedCustomLoad { id: string; name: string; watts: number; surge: number; note: string }
export type ParseResult = { ok: true; item: ParsedCustomLoad } | { ok: false; error: string }

const ID_PREFIX = 'custom'

function num(v: string): number {
  const n = Number(v)
  return Number.isFinite(n) ? n : 0
}

function nextId(existingIds: readonly string[]): string {
  let n = 1
  while (existingIds.includes(`${ID_PREFIX}${n}`)) n++
  return `${ID_PREFIX}${n}`
}

export function parseCustom(form: CustomForm, existingIds: readonly string[]): ParseResult {
  const name = form.name.trim()
  const isVa = form.mode === 'va'
  const volts = num(form.volts)
  const amps = num(form.amps)
  const computed = isVa ? volts * amps : num(form.watts)
  if (!name) return { ok: false, error: 'Give the appliance a name.' }
  if (!(computed > 0)) {
    return { ok: false, error: isVa ? 'Enter volts and amps above zero.' : 'Enter running watts above zero.' }
  }
  const surge = num(form.surge)
  if (surge > 0 && surge < computed) return { ok: false, error: 'Surge cannot be below running watts.' }
  const watts = Math.round(computed)
  return {
    ok: true,
    item: {
      id: nextId(existingIds),
      name,
      watts,
      surge: Math.round(surge > 0 ? surge : computed),
      note: isVa ? `Custom · ${volts} V × ${amps} A` : `Custom · ${watts} W`,
    },
  }
}
```

```ts
function memoryStorage(initial: Record<string, string> = {}) {
  const data = { ...initial }
  return {
    getItem: (k: string) => data[k] ?? null,
    setItem: (k: string, v: string) => { data[k] = v },
    data,
  }
}

  const s = initialStore()

  expect(loadStore(mem)).toEqual(s)
```

```ts
const HOURS_MIN = 1
const HOURS_MAX = 72

export const CustomLoadSchema = z.object({
  id: z.string().min(1),
  name: z.string().min(1).max(80),
  watts: z.number().positive(),
  surge: z.number().positive(),
  note: z.string(),
})

export const ProjectSchema = z.object({
  id: z.string().min(1),
  name: z.string().min(1).max(60),
  qty: z.record(z.string(), z.number().int().min(0).max(SIZING.MAX_QTY)),
  custom: z.array(CustomLoadSchema),
  hours: z.number().int().min(HOURS_MIN).max(HOURS_MAX),
  volts: z.union([z.literal(12), z.literal(24), z.literal(36), z.literal(48)]),
  chemistry: z.enum(['lifepo4', 'agm']),
})

export const StoreSchema = z.object({
  version: z.literal(1),
  projects: z.array(ProjectSchema).min(1),
  currentId: z.string().min(1),
})

export type CustomLoad = z.infer<typeof CustomLoadSchema>
export type Project = z.infer<typeof ProjectSchema>
export type Store = z.infer<typeof StoreSchema>
```

```ts
import { SIZING } from './catalog'
import type { CustomLoad, Project, Store } from './schema'

export const DEFAULT_QTY: Record<string, number> = { fridge: 1, lights: 1, router: 1, tv: 1, phone: 2, laptop: 1 }
const DEFAULT_HOURS = 10
const ID_PREFIX = 'p'

function nextId(projects: readonly Project[]): string {
  const max = projects.reduce((m, p) => Math.max(m, Number(p.id.slice(ID_PREFIX.length)) || 0), 0)
  return `${ID_PREFIX}${max + 1}`
}

function newProject(projects: readonly Project[], name?: string): Project {
  const id = nextId(projects)
  return { id, name: name ?? `Project ${id.slice(ID_PREFIX.length)}`, qty: { ...DEFAULT_QTY }, custom: [], hours: DEFAULT_HOURS, volts: 48, chemistry: 'lifepo4' }
}

export function initialStore(): Store {
  const p = newProject([], 'Home')
  return { version: 1, projects: [p], currentId: p.id }
}

export function currentProject(store: Store): Project {
  return store.projects.find((p) => p.id === store.currentId) ?? store.projects[0]!
}

export function patchCurrent(store: Store, patch: Partial<Project>): Store {
  const id = currentProject(store).id
  return { ...store, projects: store.projects.map((p) => (p.id === id ? { ...p, ...patch } : p)) }
}

export function addProject(store: Store, name?: string): Store {
  const p = newProject(store.projects, name)
  return { ...store, projects: [...store.projects, p], currentId: p.id }
}

export function duplicateProject(store: Store): Store {
  const src = currentProject(store)
  const copy: Project = { ...src, id: nextId(store.projects), name: `${src.name} copy`, qty: { ...src.qty }, custom: [...src.custom] }
  return { ...store, projects: [...store.projects, copy], currentId: copy.id }
}

export function deleteProject(store: Store): Store {
  if (store.projects.length <= 1) return store
  const id = currentProject(store).id
  const rest = store.projects.filter((p) => p.id !== id)
  return { ...store, projects: rest, currentId: rest[0]!.id }
}

export function selectProject(store: Store, id: string): Store {
  return store.projects.some((p) => p.id === id) ? { ...store, currentId: id } : store
}

export function setQty(store: Store, itemId: string, n: number): Store {
  const p = currentProject(store)
  const qty = { ...p.qty, [itemId]: Math.max(0, Math.min(SIZING.MAX_QTY, Math.trunc(n))) }
  return patchCurrent(store, { qty })
}

export function addCustom(store: Store, item: CustomLoad): Store {
  const p = currentProject(store)
  return patchCurrent(store, { custom: [...p.custom, item], qty: { ...p.qty, [item.id]: 1 } })
}

export function removeCustom(store: Store, itemId: string): Store {
  const p = currentProject(store)
  const { [itemId]: _dropped, ...qty } = p.qty
  return patchCurrent(store, { custom: p.custom.filter((c) => c.id !== itemId), qty })
}
```

```ts
import { StoreSchema, type Store } from './schema'

export const STORAGE_KEY = 'solar-dimension.projects.v1'

type Reader = Pick<Storage, 'getItem'>
type Writer = Pick<Storage, 'setItem'>

function browserStorage(): Storage | null {
  return typeof window !== 'undefined' && window.localStorage ? window.localStorage : null
}

export function loadStore(storage: Reader | null = browserStorage()): Store | null {
  if (!storage) return null
  try {
    const raw = storage.getItem(STORAGE_KEY)
    if (!raw) return null
    const parsed = StoreSchema.safeParse(JSON.parse(raw))
    return parsed.success ? parsed.data : null
  } catch {
    return null
  }
}

export function saveStore(store: Store, storage: Writer | null = browserStorage()): void {
  if (!storage) return
  try {
    storage.setItem(STORAGE_KEY, JSON.stringify(store))
  } catch {
    // Private mode or blocked storage: the in-memory store keeps working.
  }
}
```

```ts
import { useCallback, useEffect, useState } from 'react'
import { loadStore, saveStore } from '../lib/persistence'
import type { Project, Store } from '../lib/schema'
import { currentProject, initialStore } from '../lib/store'

export function useProjectStore(): { store: Store; project: Project; commit: (next: Store) => void } {
  const [store, setStore] = useState<Store>(initialStore)

  useEffect(() => {
    const loaded = loadStore()
    if (loaded) setStore(loaded)
  }, [])

  const commit = useCallback((next: Store) => {
    setStore(next)
    saveStore(next)
  }, [])

  return { store, project: currentProject(store), commit }
}
```

```tsx
import { createFileRoute, useNavigate } from '@tanstack/react-router'
import { ProjectBar } from '../components/ProjectBar'
import { Stepper, type Step } from '../components/Stepper'
import { WizardFooter } from '../components/WizardFooter'
import { useProjectStore } from '../hooks/useProjectStore'
import { fmtKwh, fmtW } from '../lib/format'
import { compute, type SizingResult } from '../lib/sizing'

interface Search { step: Step }

function parseStep(v: unknown): Step {
  return v === 2 || v === '2' ? 2 : v === 3 || v === '3' ? 3 : 1
}

export const Route = createFileRoute('/')({
  validateSearch: (search: Record<string, unknown>): Search => ({ step: parseStep(search.step) }),
  component: KitBuilderPage,
})

function footerSummary(step: Step, projectName: string, r: SizingResult): string {
  if (step === 1) return `${projectName} · ${r.count} appliances · ${fmtW(r.runningW)} running · ${fmtW(r.surgeW)} peak`
  if (step === 2) return `${r.hours} h at ${r.volts} V · ${fmtKwh(r.loadWh)} of load`
  if (r.empty || !r.battery) return 'Nothing selected yet'
  return `${r.battery.count} batteries · ${fmtW(r.inverter.w)} inverter · ${r.cable.gauge}`
}

function KitBuilderPage() {
  const { step } = Route.useSearch()
  const navigate = useNavigate({ from: '/' })
  const { store, project, commit } = useProjectStore()
  const result = compute({ qty: project.qty, custom: project.custom, hours: project.hours, volts: project.volts, chemistry: project.chemistry })
  const go = (next: Step) => navigate({ search: { step: next } })

  return (
    <main className="page">
      <header className="header">
        <div className="brand">Kit Builder</div>
        <Stepper step={step} />
      </header>
      <ProjectBar store={store} commit={commit} />
      <section className="stage">
        {step === 1 && <h2>Which appliances need to stay on?</h2>}
        {step === 2 && <h2>How long should the backup last?</h2>}
        {step === 3 && <h2>Kit for {project.name}</h2>}
      </section>
      <WizardFooter
        step={step}
        summary={footerSummary(step, project.name, result)}
        onBack={() => go((step - 1) as Step)}
        onNext={() => go((step + 1) as Step)}
        onStartOver={() => go(1)}
      />
    </main>
  )
}
```

```tsx
type Filter = 'All' | Category

export function ApplianceGrid({ store, commit }: { store: Store; commit: (s: Store) => void }) {
  const [filter, setFilter] = useState<Filter>('All')
  const [query, setQuery] = useState('')
  const project = currentProject(store)
  const needle = query.trim().toLowerCase()
  const items = catalog(project.custom).filter(
    (a) => (filter === 'All' || a.cat === filter) && (!needle || a.name.toLowerCase().includes(needle)),
  )
  const filters: Filter[] = ['All', ...CATEGORIES]

  return (
    <div className="grid-block">
      <div className="toolbar">
        <p className="muted">Add each one and set how many. Running watts drive the battery, surge watts drive the inverter.</p>

      </div>
      <div className="hs">
        {filters.map((f) => (
          <button key={f} type="button" className={`chip${f === filter ? ' chip--on' : ''}`} onClick={() => setFilter(f)}>{f}</button>
        ))}
      </div>
      <div className="grid">
        {items.map((item) => {
```

```tsx
import { useState } from 'react'
import { EMPTY_FORM, parseCustom, type CustomForm } from '../lib/custom'
import type { Store } from '../lib/schema'
import { addCustom, currentProject } from '../lib/store'

export function CustomApplianceForm({ store, commit }: { store: Store; commit: (s: Store) => void }) {
  const [form, setForm] = useState<CustomForm>(EMPTY_FORM)
  const [error, setError] = useState('')
  const set = (patch: Partial<CustomForm>) => { setForm({ ...form, ...patch }); setError('') }

  function submit() {
    const res = parseCustom(form, currentProject(store).custom.map((c) => c.id))
    if (!res.ok) { setError(res.error); return }
    commit(addCustom(store, res.item))
    setForm(EMPTY_FORM)
  }

  const isVa = form.mode === 'va'
  return (
    <div className="custom">
      <div className="custom__head">
        <div>
          <div className="custom__title">Something not on the list?</div>
          <div className="muted custom__hint">Type the watts from the label, or the volts and amps and we multiply them.</div>
        </div>
        <div className="custom__modes">
          <button type="button" className={`seg${!isVa ? ' seg--on' : ''}`} onClick={() => set({ mode: 'watts' })}>Watts</button>
          <button type="button" className={`seg${isVa ? ' seg--on' : ''}`} onClick={() => set({ mode: 'va' })}>Volts × amps</button>
        </div>
      </div>
      <div className="custom__row">
        <label className="field field--grow"><span className="lab">Name</span><input className="in" type="text" placeholder="Pool pump" value={form.name} onChange={(e) => set({ name: e.target.value })} /></label>
        {isVa ? (
          <>
            <label className="field"><span className="lab">Volts</span><input className="in" type="number" min="0" placeholder="230" value={form.volts} onChange={(e) => set({ volts: e.target.value })} /></label>

          </>
        ) : (
          <label className="field"><span className="lab">Running W</span><input className="in" type="number" min="0" placeholder="750" value={form.watts} onChange={(e) => set({ watts: e.target.value })} /></label>
        )}
        <label className="field"><span className="lab">Surge W (optional)</span><input className="in" type="number" min="0" placeholder="same as running" value={form.surge} onChange={(e) => set({ surge: e.target.value })} /></label>
        <button type="button" className="btn btn--primary custom__submit" onClick={submit}>Add appliance</button>
      </div>
      {error && <div className="error" role="alert">{error}</div>}
    </div>
  )
}
```

```tsx
import { CHEMISTRIES, VOLTAGES, VOLTAGE_HINTS, type ChemistryId, type Voltage } from '../lib/catalog'
import { fmtKwh, fmtW } from '../lib/format'
import type { Store } from '../lib/schema'
import type { SizingResult } from '../lib/sizing'
import { currentProject, patchCurrent } from '../lib/store'

const PRESET_HOURS = [4, 8, 10, 12, 24, 48] as const
const HOURS_MIN = 1
const HOURS_MAX = 48

interface Props { store: Store; commit: (s: Store) => void; result: SizingResult }

export function BackupWindow({ store, commit, result }: Props) {
  const p = currentProject(store)
  const setHours = (hours: number) => commit(patchCurrent(store, { hours }))
  const summary = result.empty
    ? 'Go back and add at least one appliance.'
    : `Your ${result.count} appliances draw ${fmtW(result.runningW)} and use about ${fmtKwh(result.loadWh)} over ${p.hours} hours. We size the bank to ${fmtKwh(result.requiredWh)} after inverter losses and depth of discharge.`

          <div className="hs presets">

              <button key={v} type="button" className={`opt${v === p.volts ? ' opt--on' : ''}`} onClick={() => commit(patchCurrent(store, { volts: v }))}>
                <span className="opt__t">{v} V</span><span className="opt__h">{VOLTAGE_HINTS[v]}</span>

      </div>
    </div>
  )
}
```

```tsx
import { CHEMISTRIES } from '../lib/catalog'
import { fmtHours, fmtInt, fmtKwh, fmtW } from '../lib/format'
import type { SizingResult } from '../lib/sizing'

export interface Part { name: string; spec: string; note: string; qty: string }

export function buildParts(r: SizingResult): Part[] {
  const b = r.battery
  if (r.empty || !b) return []
  const cm = Math.round(r.cable.linkLengthM * 100)
  const perRun = r.cable.runs > 1 ? ' per run' : ''
  return [
    { name: 'Inverter', spec: `${fmtW(r.inverter.w)} pure sine, ${r.volts} V DC input`, note: `${fmtW(r.inverter.surge)} surge · covers ${fmtW(r.runningW)} running, ${fmtW(r.surgeW)} peak`, qty: '1' },
    { name: 'Batteries', spec: b.unitLabel, note: `${b.series} in series × ${b.parallel} in parallel (${b.config}) · ${fmtKwh(b.bankWh)} bank, ${fmtKwh(b.usableWh)} usable`, qty: String(b.count) },
    { name: 'Main cable', spec: `${r.cable.gauge} (${r.cable.mm2} mm²) copper, ${r.cable.mainLengthM} m each`, note: `Battery to inverter · ${fmtInt(r.cable.amps)} A at full inverter load · ${r.cable.dropPct}% drop`, qty: String(r.cable.mainCount) },
    { name: 'Interconnects', spec: `${r.cable.gauge} jumpers, ${cm} cm each`, note: `${r.cable.seriesLinks} series links · ${r.cable.parallelLinks} parallel jumpers`, qty: String(r.cable.seriesLinks + r.cable.parallelLinks) },
    { name: 'Fuse', spec: `${r.cable.fuseA} A class T${perRun}`, note: 'Inline on the positive lead, as close to the bank as possible', qty: String(r.cable.runs) },
  ]
}

interface Props { result: SizingResult; projectName: string; onEditAppliances: () => void; onEditWindow: () => void }

export function KitSummary({ result: r, projectName, onEditAppliances, onEditWindow }: Props) {
  const parts = buildParts(r)
  const dod = Math.round(CHEMISTRIES[r.chemistry].dod * 100)
  return (
    <div className="kit">
      <div className="kit__head">
        <p className="muted">{r.empty ? 'Add appliances first.' : `Backs up ${r.count} appliances for ${r.hours} hours on a ${r.volts} V ${r.chemistryLabel} bank.`}</p>
        <div className="kit__edit">
          <button type="button" className="chip" onClick={onEditAppliances}>Edit appliances</button>
          <button type="button" className="chip" onClick={onEditWindow}>Edit window</button>
        </div>
      </div>
      <div className="stats">
        <div className="stat"><div className="lab">Running load</div><div className="stat__v num">{fmtW(r.runningW)}</div></div>
        <div className="stat"><div className="lab">Start-up peak</div><div className="stat__v num">{fmtW(r.surgeW)}</div></div>
        <div className="stat"><div className="lab">Energy needed</div><div className="stat__v num">{fmtKwh(r.requiredWh)}</div></div>
        <div className="stat stat--accent"><div className="lab">Runtime</div><div className="stat__v num">{r.battery && !r.empty ? `~${fmtHours(r.battery.hoursActual)}` : '—'}</div></div>
      </div>
      <div className="parts">
        <div className="parts__head"><div className="lab">Part</div><div className="lab">Specification</div><div className="lab parts__qty">Quantity</div></div>
        {parts.map((pt) => (
          <div key={pt.name} className="part">
            <div className="part__name">{pt.name}</div>
            <div className="part__body"><div className="part__spec">{pt.spec}</div><div className="part__note">{pt.note}</div></div>
            <div className="part__qty num">×{pt.qty}</div>
          </div>
        ))}
        {parts.length === 0 && <p className="muted">{projectName} has nothing selected yet.</p>}
      </div>
      {r.warnings.length > 0 && (
        <div className="warn" role="status">
          {r.warnings.map((w) => <div key={w}>{w}</div>)}
        </div>
      )}
      <p className="fine muted">Assumes 90% inverter efficiency, 25% inverter headroom, {dod}% depth of discharge, 1,100 W per ton of cooling and one motor starting at a time. Verify conductor sizes against local code.</p>
    </div>
  )
}
```
